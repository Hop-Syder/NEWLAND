   112→function fmtDateTime(iso: string): string {
   113→  return new Intl.DateTimeFormat("fr-FR", {
   114→    day: "numeric",
   115→    month: "short",
   116→    hour: "2-digit",
   117→    minute: "2-digit",

   121→function relDate(iso: string): string {
   122→  const days = Math.floor((Date.now() - new Date(iso).getTime()) / 86_400_000);
   123→  if (days <= 0) return "Aujourd'hui";
   124→  if (days === 1) return "Hier";
   125→  if (days < 7) return `Il y a ${days} j`;
   126→  return SHORT_DATE.format(new Date(iso));
   127→}

   133→/** CA des 7 derniers jours, jour par jour (pour le graphique en barres). */
   134→function computeLast7Days(orders: Order[]): {
   135→  label: string;
   136→  total: number;
   137→  key: string;
   138→}[] {
   139→  const out: { label: string; total: number; key: string }[] = [];
   140→  for (let i = 6; i >= 0; i--) {
   141→    const d = new Date();
   142→    d.setDate(d.getDate() - i);
   143→    const key = d.toDateString();
   144→    const total = orders
   145→      .filter(
   146→        (o) =>
   147→          statusOf(o) !== "annulee" &&
   148→          new Date(o.createdAt).toDateString() === key
   149→      )
   150→      .reduce((sum, o) => sum + o.total, 0);
   151→    out.push({
   152→      key,
   153→      total,
   154→      label: new Intl.DateTimeFormat("fr-FR", { weekday: "short" })
   155→        .format(d)
   156→        .replace(".", ""),
   157→    });
   158→  }
   159→  return out;
   160→}
   161→
   162→function slugify(name: string): string {
   163→  return name
   164→    .toLowerCase()
   165→    .normalize("NFD")
   166→    .replace(/[\u0300-\u036f]/g, "")
   167→    .replace(/[^a-z0-9]+/g, "-")
   168→    .replace(/(^-|-$)/g, "");
   169→}

   439→function DashboardTab({ onGoProducts }: { onGoProducts: () => void }) {
   440→  const adminOrders = useShop((s) => s.adminOrders);
   441→  const clients = useShop((s) => s.clients);
   442→  const products = useShop((s) => s.products);
   443→  const stock = useShop((s) => s.stock);
   444→
   445→  const active = adminOrders.filter((o) => statusOf(o) !== "annulee");
   446→  const revenue = active.reduce((sum, o) => sum + o.total, 0);
   447→  const avg = active.length ? Math.round(revenue / active.length) : 0;
   448→  const pendingCount = adminOrders.filter(
   449→    (o) => statusOf(o) === "en_attente"
   450→  ).length;
   451→  const lowStock = products.filter((p) => (stock[p.id] ?? 0) <= 3);
   452→
   453→  // Chiffre d'affaires des 7 derniers jours (barres CSS, sans librairie).
   454→  const days = computeLast7Days(adminOrders);
   455→
   456→  const maxDay = Math.max(...days.map((d) => d.total), 1);
   457→  const recent = adminOrders.slice(0, 5);
   458→
   459→  return (
   460→    <div className="grid gap-5">
   461→      <div>
   462→        <h1 className="text-xl font-extrabold sm:text-2xl">Tableau de bord</h1>

   464→          Vue d&apos;ensemble de la boutique — mise à jour en direct.

   466→      </div>

   468→      {/* Indicateurs clés */}

   471→          icon={Banknote}
   472→          label="Chiffre d'affaires"

   500→        />

   504→        {/* Graphique 7 jours */}
   505→        <section
   506→          aria-label="Chiffre d'affaires des 7 derniers jours"

   510→            CA — 7 derniers jours

   512→          <div className="mt-4 grid h-44 grid-cols-7 items-end gap-2 sm:gap-3">
   513→            {days.map((d) => (

   532→                </span>
   533→              </div>
   534→            ))}
   535→          </div>
   536→        </section>
   537→
   538→        {/* Alertes stock */}
   539→        <section
   540→          aria-label="Alertes de stock"
   541→          className="rounded-md border border-[#2a2a2a] bg-[#161616] p-5 lg:col-span-2"
   542→        >
   543→          <h2 className="flex items-center gap-2 text-sm font-extrabold uppercase tracking-wide text-[#a3a3a3]">
   544→            <AlertTriangle

   271→function LoginGate() {
   272→  const customer = useShop((s) => s.customer);
   273→  const pinStored = useShop((s) => s.pin);
   274→  const setPin = useShop((s) => s.setPin);
   275→  const unlock = useShop((s) => s.unlock);
   276→  const saveCustomer = useShop((s) => s.saveCustomer);
   277→  const resendCredentials = useShop((s) => s.resendCredentials);
   278→
   279→  const [phone, setPhone] = useState("");
   280→  const [pin, setPinField] = useState("");
   281→  const [confirm, setConfirm] = useState("");
   282→  const [email, setEmail] = useState(customer?.email ?? "");
   283→  const [show, setShow] = useState(false);
   284→  const [wipeArmed, setWipeArmed] = useState(false);
   285→
   286→  const hasPin = typeof pinStored === "string";
   287→  const lastDigits = customer
   288→    ? customer.phone.replace(/\D/g, "").slice(-4)
   289→    : "";
   290→
   291→  const submit = (e: React.FormEvent) => {
   292→    e.preventDefault();
   293→    if (!customer) return;
   294→
   295→    if (!phoneMatches(phone, customer.phone)) {
   296→      toast({
   297→        title: "Numéro inconnu",
   298→        description: "Aucun compte lié à ce numéro sur cet appareil.",
   299→        variant: "destructive",
   300→      });
   301→      return;
   302→    }
   303→
   304→    if (!hasPin) {
   305→      // Première sécurisation : le client choisit lui-même son code PIN.
   306→      if (!isValidPin(pin)) {
   307→        toast({
   308→          title: "Code PIN invalide",
   309→          description: "Choisissez exactement 4 chiffres.",
   310→          variant: "destructive",
   311→        });
   312→        return;
   313→      }
   314→      if (pin !== confirm) {
   315→        toast({
   316→          title: "Codes différents",
   317→          description: "La confirmation ne correspond pas.",
   318→          variant: "destructive",
   319→        });
   320→        return;
   321→      }
   322→      const mail = email.trim();
   323→      if (!isValidEmail(mail)) {
   324→        toast({
   325→          title: "Email requis",
   326→          description: "La copie de vos identifiants y sera envoyée.",
   327→          variant: "destructive",
   328→        });
   329→        return;
   330→      }
   331→      if (mail !== (customer.email ?? "")) {
   332→        saveCustomer({ ...customer, email: mail });
   333→      }
   334→      setPin(pin);
   335→      toast({
   336→        title: "Compte sécurisé",
   337→        description: `Copie des identifiants envoyée à ${mail}.`,
   338→      });
   339→      return;
   340→    }
   341→
   342→    if (pin !== pinStored) {
   343→      toast({
   344→        title: "Code PIN incorrect",
   345→        description: "Réessayez ou demandez une copie par email.",
   346→        variant: "destructive",
   347→      });
   348→      return;
   349→    }
   350→
   351→    unlock();
   352→    toast({
   353→      title: `Bienvenue ${customer.name.split(" ")[0]} !`,
   354→      description: "Votre espace client est déverrouillé.",
   355→    });
   356→  };
   357→
   358→  const forgot = () => {
   359→    const mail = resendCredentials();
   360→    if (mail) {
   361→      toast({
   362→        title: "Copie envoyée",
   363→        description: `Vos identifiants ont été renvoyés à ${mail}.`,
   364→      });
   365→    } else {
   366→      toast({
   367→        title: "Aucun email enregistré",
   368→        description: "Effacez ce compte de cet appareil, puis recommandez.",
   369→      });
   370→    }
   371→  };
   372→
   373→  const wipe = () => {
   374→    try {
   375→      useShop.persist.clearStorage();
   376→    } catch {
   377→      /* stockage indisponible : on recharge quand même */
   378→    }
   379→    window.location.reload();
   380→  };
   381→
   382→  return (
   383→    <div className="mx-auto w-full max-w-md px-4 py-12 lg:py-16">
   384→      <Reveal>
   385→        <div className="rounded-md border border-[#e6e3db] bg-white p-6 shadow-sm sm:p-8">
   386→          <div className="text-center">
   387→            <span className="mx-auto flex h-14 w-14 items-center justify-center rounded-md bg-[#141414]">
   388→              <LockKeyhole className="h-7 w-7 text-[#d4af37]" aria-hidden="true" />
   389→            </span>
   390→            <h1 className="mt-4 text-2xl font-extrabold text-[#0b0b0b]">
   391→              Espace client sécurisé
   392→            </h1>
   393→            <p className="mt-1.5 text-sm text-muted-foreground">
   394→              {hasPin
   395→                ? "Identifiez-vous avec votre numéro et votre code PIN à 4 chiffres."
   396→                : "Confirmez votre numéro, puis choisissez votre code PIN à 4 chiffres."}
   397→            </p>
   398→            {lastDigits && (
   399→              <p className="mt-1 text-xs text-muted-foreground">
   400→                Compte enregistré se terminant par {lastDigits}
   401→              </p>
   402→            )}
   403→          </div>
   404→
   405→          <form onSubmit={submit} className="mt-6 grid gap-4">
   406→            <div className="grid gap-2">
   407→              <Label htmlFor="lg-phone">Téléphone (identifiant) *</Label>
   408→              <Input
   409→                id="lg-phone"
   410→                type="tel"
   411→                value={phone}
   412→                onChange={(e) => setPhone(e.target.value)}
   413→                placeholder="+229 01 97 00 00 00"
   414→                autoComplete="username"
   415→                className="h-11 rounded-sm border-[#d8d4ca]"
   416→              />
   417→            </div>
   418→
   419→            <div className="grid gap-2">
   420→              <Label htmlFor="lg-pin">
   421→                {hasPin ? "Code PIN *" : "Nouveau code PIN (4 chiffres) *"}
   422→              </Label>
   423→              <div className="relative">
   424→                <Input
   425→                  id="lg-pin"
   426→                  type={show ? "text" : "password"}
   427→                  inputMode="numeric"
   428→                  maxLength={4}
   429→                  value={pin}
   430→                  onChange={(e) => setPinField(e.target.value.replace(/\D/g, ""))}
   431→                  autoComplete={hasPin ? "current-password" : "new-password"}
   432→                  className="h-11 rounded-sm border-[#d8d4ca] pr-11 tracking-[0.5em]"
   433→                />
   434→                <button
   435→                  type="button"
   436→                  aria-label={show ? "Masquer le code" : "Afficher le code"}
   437→                  onClick={() => setShow((v) => !v)}
   438→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-[#8f8f8f] hover:text-[#111111]"
   439→                >
   440→                  {show ? (
   441→                    <EyeOff className="h-4 w-4" aria-hidden="true" />
   442→                  ) : (
   443→                    <Eye className="h-4 w-4" aria-hidden="true" />
   444→                  )}
   445→                </button>
   446→              </div>
   447→            </div>
   448→
   449→            {!hasPin && (
   450→              <>
   451→                <div className="grid gap-2">
   452→                  <Label htmlFor="lg-confirm">Confirmer le code PIN *</Label>
   453→                  <Input
   454→                    id="lg-confirm"
   455→                    type="password"
   456→                    inputMode="numeric"
   457→                    maxLength={4}
   458→                    value={confirm}
   459→                    onChange={(e) =>
   460→                      setConfirm(e.target.value.replace(/\D/g, ""))
   461→                    }
   462→                    autoComplete="new-password"
   463→                    className="h-11 rounded-sm border-[#d8d4ca] tracking-[0.5em]"
   464→                  />
   465→                </div>
   466→                <div className="grid gap-2">
   467→                  <Label htmlFor="lg-email">Email de réception *</Label>
   468→                  <Input
   469→                    id="lg-email"
   470→                    type="email"
   471→                    value={email}
   472→                    onChange={(e) => setEmail(e.target.value)}
   473→                    placeholder="vous@exemple.com"
   474→                    autoComplete="email"
   475→                    className="h-11 rounded-sm border-[#d8d4ca]"
   476→                  />
   477→                  <p className="text-xs text-muted-foreground">
   478→                    Une copie de vos identifiants (numéro + code PIN) y sera
   479→                    envoyée.
   480→                  </p>
   481→                </div>
   482→              </>
   483→            )}
   484→
   485→            <Button
   486→              type="submit"
   487→              size="lg"
   488→              className="h-12 rounded-sm bg-[#d4af37] font-extrabold text-[#0b0b0b] hover:bg-[#bfa032]"
   489→            >
   490→              {hasPin ? "Déverrouiller mon espace" : "Sécuriser mon compte"}
   491→            </Button>
   492→          </form>
   493→
   494→          {hasPin && (
   495→            <button
   496→              type="button"
   497→              onClick={forgot}
   498→              className="mt-3 w-full text-center text-sm font-bold text-[#8a6a14] underline-offset-2 hover:underline"
   499→            >
   500→              Identifiants oubliés ? Renvoyer une copie par email
   501→            </button>
   502→          )}
   503→
   504→          <Separator className="my-5" />
   505→
   506→          <p className="text-center text-xs text-muted-foreground">
   507→            Ce n&apos;est pas votre compte ?{" "}
   508→            {wipeArmed ? (
   509→              <button
   510→                type="button"
   511→                onClick={wipe}
   512→                className="font-bold text-[#c6482f] underline"

   514→                Confirmer l&apos;effacement

   525→          </p>
